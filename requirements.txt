# Tier 1 environment: everything `python run_tier1.py` needs.
#
# These four packages are enough to recompute the held-out species score, every agreement
# statistic, and every figure that is drawn from the committed count tables. No model, no
# video library, no graphics card.
#
#   python -m venv .venv
#   .venv\Scripts\activate            (Windows)      source .venv/bin/activate   (macOS / Linux)
#   pip install -r requirements.txt
#   python run_tier1.py
#
# Python 3.11 or newer. The versions are pinned to the ones that produced the published
# tables and figures. Newer versions are expected to give the same numbers; the figure files
# may then differ in the last pixel (run_tier1.py reports that as a NOTE, not a failure).
#
# Tier 2 (tracking, detection, training) uses two further environments:
#   requirements-track.txt   tracking, counting from stored detections, photo figures
#   requirements-deim.txt    DEIMv2 detection and training (GPU)
numpy==2.4.4
pandas==2.3.3
scipy==1.15.3
matplotlib==3.10.8
